            expense("Arriendo", 380000L, "Vivienda", false),
            expense("Bip", 42000L, "Transporte", false)
        )
    }

package cl.finantzas.app.data.local.dao

import androidx.room.*
import cl.finantzas.app.data.local.entity.ExpenseEntityClp
import kotlinx.coroutines.flow.Flow

@Dao
interface ExpenseDao {
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insert(expense: ExpenseEntityClp)

    @Update
    suspend fun update(expense: ExpenseEntityClp)

    @Delete
    suspend fun delete(expense: ExpenseEntityClp)

    @Query("SELECT * FROM expenses WHERE user_id = :userId ORDER BY created_at DESC")
    fun getExpensesByUser(userId: String): Flow<List<ExpenseEntityClp>>

    @Query("SELECT * FROM expenses WHERE id = :expenseId AND user_id = :userId LIMIT 1")
    suspend fun getExpenseById(expenseId: String, userId: String): ExpenseEntityClp?

    @Query("SELECT COALESCE(SUM(monthly_impact_clp), 0) FROM expenses WHERE user_id = :userId")
    fun getTotalMonthlyImpactClp(userId: String): Flow<Long>

    @Query("DELETE FROM expenses WHERE user_id = :userId AND id LIKE 'demo-%'")
    suspend fun deleteDemoExpenses(userId: String)

    @Transaction
    suspend fun restoreDemoExpenses(userId: String, demoExpenses: List<ExpenseEntityClp>) {
        deleteDemoExpenses(userId)
        demoExpenses.forEach { insert(it) }
    }
    @Query("SELECT * FROM expenses WHERE user_id = :userId ORDER BY created_at ASC")
    suspend fun getExpensesForSync(userId: String): List<ExpenseEntityClp>
}

import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column

import androidx.compose.material3.Text

import androidx.compose.runtime.getValue

import androidx.compose.runtime.remember
import androidx.compose.runtime.rememberCoroutineScope
import androidx.compose.runtime.setValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.unit.dp
import cl.finantzas.app.data.demo.ExpenseDemoDataClp
import cl.finantzas.app.data.local.database.SecureDatabaseProvider
import cl.finantzas.app.data.repository.RoomExpenseRepository
import cl.finantzas.app.ui.viewmodel.ExpenseViewModel
import kotlinx.coroutines.launch

private const val DEMO_USER_ID = "finantzas-demo-local"

@Composable
fun ExpenseDemoScreen() {
    val context = LocalContext.current.applicationContext
    val dao = remember(context) { SecureDatabaseProvider.getDatabase(context).expenseDao() }
    val repository = remember(dao) { RoomExpenseRepository(dao) }
    val viewModel = remember(repository) { ExpenseViewModel(repository, DEMO_USER_ID) }
    val expenses by viewModel.expenses.collectAsState()
    val totalMonthly by viewModel.totalMonthlyClp.collectAsState()
    val scope = rememberCoroutineScope()
    var restoring by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }
    val antMonthly = expenses.filter { it.isAntExpense }.sumOf { it.monthlyImpactClp }

    Column(verticalArrangement = Arrangement.spacedBy(16.dp)) {
        KpiCards(
            gastoMensual = totalMonthly,
            proyeccionAnual = totalMonthly * 12L,
            gastoHormiga = antMonthly,
            ahorroPotencial = antMonthly * 12L
        )
        Button(
            onClick = {
                if (!restoring) {
                    restoring = true
                    error = null
                    scope.launch {
                        try {
                            dao.restoreDemoExpenses(DEMO_USER_ID, ExpenseDemoDataClp.create(DEMO_USER_ID))
                        } catch (e: Exception) {
                            error = "No se pudieron restaurar los datos: ${e.message ?: "error desconocido"}"
                        } finally {
                            restoring = false
                        }
                    }
                }
            },
            enabled = !restoring,
            modifier = Modifier.fillMaxWidth()
        ) { Text(if (restoring) "Restaurando..." else "Restaurar datos de ejemplo") }
        if (error != null) Text(error!!)
    }
}

@Composable
fun FinANTzasNavigation() {
    val navController = rememberNavController()
    val entradaActual = navController.currentBackStackEntryAsState()
    val rutaActual = entradaActual.value?.destination?.route ?: "registro"

    Scaffold(
        modifier = Modifier.fillMaxSize(),
        topBar = {
            EncabezadoFinANTzas()
        },
        bottomBar = {
            BarraNavegacion(
                rutaActual = rutaActual,
                onNavigate = { ruta ->
                    navController.navigate(ruta) {
                        popUpTo("registro") {
                            saveState = true
                        }
                        launchSingleTop = true
                        restoreState = true
                    }
                }
            )
        }
    ) { innerPadding ->
        NavHost(
            navController = navController,
            startDestination = "registro",
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .padding(16.dp)
        ) {
            composable("registro") {
                PantallaRegistro()
            }
            composable("graficos") {
                PantallaTemporal("Sankey, Anillos y % de Fuga")
            }
            composable("simulador") {
                PantallaTemporal("Simulador de Ahorro")
            }

            }

@Composable
private fun BarraNavegacion(
    rutaActual: String,
    onNavigate: (String) -> Unit
) {
    NavigationBar(
        containerColor = MaterialTheme.colorScheme.surface,
        contentColor = Petrol
    ) {
        pantallas.forEach { pantalla ->
            val seleccionada = rutaActual == pantalla.ruta

            ItemNavegacion(
                pantalla = pantalla,
                seleccionada = seleccionada,
                onClick = {
                    if (!seleccionada) {
                        onNavigate(pantalla.ruta)
                    }
                }
            )
        }
    }
}

@Composable

    pantalla: Pantalla,
    seleccionada: Boolean,
    onClick: () -> Unit
) {
    NavigationBarItem(
        selected = seleccionada,
        onClick = onClick,
        icon = {
            Icon(
                imageVector = pantalla.icono,
                contentDescription = pantalla.titulo
            )
        },
        label = {

                maxLines = 1,
                fontWeight = if (seleccionada) {
                    FontWeight.Bold
                } else {
                    FontWeight.Normal

            )

        alwaysShowLabel = true,
        colors = NavigationBarItemDefaults.colors(
            selectedIconColor = Color.White,
            selectedTextColor = Petrol,

package cl.finantzas.app.ui.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import cl.finantzas.app.data.local.entity.ExpenseEntityClp
import cl.finantzas.app.data.repository.ExpenseReaderRepository
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.distinctUntilChanged
import kotlinx.coroutines.flow.stateIn

class ExpenseViewModel(
    repository: ExpenseReaderRepository,
    userId: String
) : ViewModel() {

    val expenses: StateFlow<List<ExpenseEntityClp>> =
        repository.getExpensesByUser(userId)
            .distinctUntilChanged()
            .stateIn(
                scope = viewModelScope,
                started = SharingStarted.WhileSubscribed(5000),
                initialValue = emptyList()
            )

    val totalMonthlyClp: StateFlow<Long> =
        repository.getTotalMonthlyImpactClp(userId)
            .distinctUntilChanged()
            .stateIn(
                scope = viewModelScope,
                started = SharingStarted.WhileSubscribed(5000),
                initialValue = 0L
            )
}
